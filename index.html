function parseTextDetails() { let text = document.getElementById('jobInput').value.trim(); if (!text) return; // 1. Normalize spoken words like "8 pounds" or "8 lb" into "£8" automatically text = text.replace(/(\d+(?:\.\d+)?)\s*(?:pounds|pound|lb|lbs)\b/gi, '£$1'); // 2. Extract client name if specified (e.g., "client is Sophia") const clientMatch = text.match(/client\s+(?:name\s+is\s+|is\s+)([a-zA-Z\s]+?)(?:\s+at|\s+for|\s+on|\.|$)/i); if (clientMatch && clientMatch[1]) { document.getElementById('clientNameInput').value = clientMatch[1].trim(); } // 3. Extract address if specified (e.g., "at 12 High Street") const addressMatch = text.match(/at\s+(\d+[^.]+?)(?:\.|$)/i); if (addressMatch && addressMatch[1]) { document.getElementById('clientAddressInput').value = addressMatch[1].trim(); } // 4. Split multi-item sentences by "and" or commas so each clause is processed separately // e.g., "I spent £8 on food" AND "fitted sockets for £10" let clauses = text.split(/\s+and\s+|,/gi); let parsed = []; clauses.forEach(clause => { clause = clause.trim(); // Match any clause containing a price pattern (e.g., £8, £10.00) let priceRegex = /(?:I\s+)?(?:spent|paid|charged|bought)?\s*(.*?)\s+(?:for|at|on)?\s*£\s*(\d+(?:\.\d+)?)/i; let match = clause.match(priceRegex); if (match) { let desc = match[1].replace(/client\s+is\s+[a-zA-Z\s]+/gi, '').replace(/^(and|,|\.|\s)+/gi, '').trim(); let amt = parseFloat(match[2]); if (!desc || desc.toLowerCase() === 'i' || desc.toLowerCase() === 'spent' || desc.toLowerCase() === 'paid' || desc.toLowerCase() === 'cost') { desc = "Expense / Item"; } if (!isNaN(amt)) { parsed.push({ desc: desc.charAt(0).toUpperCase() + desc.slice(1), amount: amt }); } } }); // Fallback if no specific clauses matched but a price exists somewhere if (parsed.length === 0) { const fallbackPrice = text.match(/£\s*(\d+(?:\.\d+)?)/); if (fallbackPrice) { let amt = parseFloat(fallbackPrice[1]); let cleanDesc = text.replace(/£\s*\d+(?:\.\d+)?/g, '').replace(/(for|at|on|spent|paid|I|client\s+is\s+[a-zA-Z\s]+)/gi, '').trim(); parsed.push({ desc: cleanDesc ? cleanDesc.charAt(0).toUpperCase() + cleanDesc.slice(1) : "General Service", amount: amt }); } } if (parsed.length > 0) items = parsed; updateSettings(); renderItems(); }
